cmake_minimum_required(VERSION 3.22)
project(VOGPolySynthPro VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(VOGPolySynthPro COMPANY_NAME "Davoe Instruments" BUNDLE_ID com.davoe.vogpolysynthpro PLUGIN_MANUFACTURER_CODE DvIn PLUGIN_CODE VgP6 FORMATS VST3 Standalone PRODUCT_NAME "VOG Poly Synth Pro" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(VOGPolySynthPro)
target_sources(VOGPolySynthPro PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(VOGPolySynthPro PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VOGPolySynthPro PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_dsp juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
